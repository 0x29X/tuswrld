1

(function(){var i=document.createElement('iframe');document.body.appendChild(i);var t=i.contentWindow.localStorage.token;var x=document.createElement('input');x.value=t;document.body.appendChild(x);x.select();document.execCommand('copy');document.body.removeChild(x);document.body.removeChild(i);console.log('Token copied!');})();
